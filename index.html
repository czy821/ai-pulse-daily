name: AI Pulse Daily 07:00 Update
on:
  schedule:
    # UTC 23:00 = 北京时间次日早上 07:00
    - cron: '0 23 * * *'
  workflow_dispatch:

jobs:
  build-briefing:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - run: pip install feedparser requests openai
      - name: Fetch 6 sources & generate 20-30 min briefing
        env:
          LLM_API_KEY: ${{ secrets.LLM_API_KEY }}
        run: python scripts/fetch_and_summarize.py
      - name: Commit daily_briefing.json
        run: |
          git config user.name "AI Pulse Bot"
          git config user.email "bot@aipulse.local"
          git add data/daily_briefing.json
          git commit -m "chore: update AI briefing 07:00" || exit 0
          git push
